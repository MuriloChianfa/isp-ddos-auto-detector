\section[Introdução e Motivação]{Introdução e Motivação}

\begin{frame}{Contexto}
\begin{itemize}
  \setlength{\itemsep}{0.55em}
  \item Ataques DDoS continuam sendo uma grande ameaça em redes de grande porte que muitas vezes sustentam ASNs menores.
  \item \textbf{Provedores de Trânsito (ITPs)} concentram políticas de roteamento e grande volume de tráfego, sendo alvo e, ao mesmo tempo, ponto de observação/mitigação (p.\,ex.\ engenharia de tráfego, BGP, RTBH, desvios para \emph{scrubbing}).
  \item Cenários acima de \textbf{100~Gbps} impõem restrições operacionais: pouco rótulo, tráfego massivo e necessidade de cooperação entre downstream e upstream [Sriram and Montgomery, 2019].
\end{itemize}
\end{frame}

\begin{frame}{Objetivos}
\begin{itemize}
  \setlength{\itemsep}{0.55em}
  \item Avaliar \textbf{quatro detectores \emph{one-class}} em agregados de fluxo, com dados reais de ITP durante ataques DDoS confirmados.%: \textbf{Autoencoder (AE)}, \textbf{Isolation Forest (IF)}, \textbf{Local Outlier Factor (LOF)} e \textbf{One-Class SVM (OCSVM)}.
  \item Três conjuntos operacionais (\textbf{DS1}--\textbf{DS3}), altamente \textbf{desequilibrados} (fração mínima de ataques no teste).
  %\item \alert{Objetivo:} Buscar um detector que apresente o melhor desempenho global e maior estabilidade entre cenário. \textbf{Janelas curtas} que favorecem detecção mais rápida, \textbf{Seleção de features} agressiva mantém qualidade com menos dimensionalidade.
  \item \alert{Objetivo:} Buscar um detector que apresente o melhor desempenho global e maior estabilidade entre cenários.
\end{itemize}
\end{frame}

\begin{frame}[t]{Datasets operacionais (DS1--DS3)}
\vspace{-0.25em}
\footnotesize
\begin{itemize}
  \setlength{\itemsep}{0.28em}
  \item \textbf{DS1}: HTTP flood em um enlace de downstream \(\sim\)10~Gb/s.
  \item \textbf{DS2}: UDP multi-vetor (\emph{carpet bombing}) com pico da ordem de 100~Gb/s, volume de fluxos muito maior.
  \item \textbf{DS3}: Ataques de \textbf{SYN-ACK} com indisponibilidade prolongada no cliente. Observado na borda do ITP por quase 2 horas.
\end{itemize}

\section[Conclusão e Perspectivas]{Conclusão e Perspectivas}

\begin{frame}{Conclusão e limitações}
\begin{itemize}
  \setlength{\itemsep}{0.55em}
  \item \textbf{Conclusão:} detecção não supervisionada por fluxo em ITP é viável com \textbf{AE} como candidato com as melhores métricas para cenários distintos até mesmo com poucas features.
  \item \textbf{Limitações:} Carece de estudo de \textbf{\emph{concept drift}} e necessidade de re-treinamento. Custos de latência, CPU e memória não foram todos quantificados.
  \item \textbf{Futuro:} validação em produção com aprendizado \textbf{online} e análise de custo operacional ponta a ponta.
\end{itemize}
\end{frame}
